#[tokio::test]
async fn generated_project_uses_real_http_and_mongodb() {
    let _ = rustls::crypto::ring::default_provider().install_default();
    let uri = std::env::var("TRI_TEST_MONGODB_URI")
        .expect("set TRI_TEST_MONGODB_URI to a test replica set");
    let name = format!("tri_generated_test_{}", uuid::Uuid::new_v4().simple());
    let database = mongodb::Client::with_uri_str(uri)
        .await
        .unwrap()
        .database(&name);
    let schema = generated_admin::schema();
    let resource = schema.resource.clone();
    let config = tri_resource_runtime::Config::new(
        "isolated-http-test-key-at-least-32-bytes".into(),
        "http://localhost:5173".into(),
    )
    .unwrap();
    let state = tri_resource_runtime::State::new(database.clone(), schema, config).unwrap();
    state.migrate().await.unwrap();
    let password = format!("fixture-{}", uuid::Uuid::new_v4());
    state
        .provision_admin("owner", &password, true, true)
        .await
        .unwrap();
    let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
    let origin = format!("http://{}", listener.local_addr().unwrap());
    let (stop, stopped) = tokio::sync::oneshot::channel::<()>();
    let server = tokio::spawn(async move {
        axum::serve(
            listener,
            tri_resource_runtime::router(state)
                .into_make_service_with_connect_info::<std::net::SocketAddr>(),
        )
        .with_graceful_shutdown(async {
            let _ = stopped.await;
        })
        .await
        .unwrap();
    });
    let client = reqwest::Client::builder()
        .no_proxy()
        .timeout(std::time::Duration::from_secs(10))
        .build()
        .unwrap();
    let signed = client
        .post(format!("{origin}/api/v1/auth/login"))
        .header("content-type", "application/json")
        .body(serde_json::json!({"username":"owner","password":password}).to_string())
        .send()
        .await
        .unwrap();
    assert_eq!(signed.status(), 200);
    let signed: serde_json::Value = serde_json::from_slice(&signed.bytes().await.unwrap()).unwrap();
    let token = signed["data"]["accessToken"].as_str().unwrap();
    let values: serde_json::Value =
        serde_json::from_str(include_str!("../../sample.json")).unwrap();
    let created = client
        .post(format!("{origin}/api/v1/{resource}"))
        .bearer_auth(token)
        .header("content-type", "application/json")
        .body(serde_json::json!({"values":values}).to_string())
        .send()
        .await
        .unwrap();
    assert_eq!(created.status(), 201);
    let created: serde_json::Value =
        serde_json::from_slice(&created.bytes().await.unwrap()).unwrap();
    assert_eq!(created["data"]["values"], values);
    let id = created["data"]["id"].as_str().unwrap();
    let updated = client
        .patch(format!("{origin}/api/v1/{resource}/{id}"))
        .bearer_auth(token)
        .header("content-type", "application/json")
        .body(serde_json::json!({"version":1,"values":values}).to_string())
        .send()
        .await
        .unwrap();
    assert_eq!(updated.status(), 200);
    assert_eq!(
        client
            .delete(format!("{origin}/api/v1/{resource}/{id}?version=1"))
            .bearer_auth(token)
            .send()
            .await
            .unwrap()
            .status(),
        409
    );
    assert_eq!(
        client
            .delete(format!("{origin}/api/v1/{resource}/{id}?version=2"))
            .bearer_auth(token)
            .send()
            .await
            .unwrap()
            .status(),
        204
    );
    assert_eq!(
        client
            .get(format!("{origin}/api/v1/{resource}/{id}"))
            .bearer_auth(token)
            .send()
            .await
            .unwrap()
            .status(),
        404
    );
    assert_eq!(
        client
            .post(format!("{origin}/api/v1/auth/logout"))
            .bearer_auth(token)
            .header("content-type", "application/json")
            .body("{}")
            .send()
            .await
            .unwrap()
            .status(),
        204
    );
    assert_eq!(
        client
            .get(format!("{origin}/api/v1/{resource}"))
            .bearer_auth(token)
            .send()
            .await
            .unwrap()
            .status(),
        401
    );
    stop.send(()).unwrap();
    server.await.unwrap();
    assert!(name.starts_with("tri_generated_test_"));
    database.drop().await.unwrap();
}
